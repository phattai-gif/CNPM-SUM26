\subsubsection{Installation Instruction}
\label{subsubsec:installation-instruction}

\paragraph*{2.2.1 Backend}
\addcontentsline{toc}{paragraph}{\protect\numberline{2.2.1}Backend}

\begin{itemize}[leftmargin=1.5em, nosep]
    \item \textbf{Step 1:} Download \& Install Visual Studio Code: Visual Studio Code.
    \item \textbf{Step 2:} Download \& Install Git: Git.
    \item \textbf{Step 3:} Download \& Install Python: Python.
    \item \textbf{Step 4:} Download \& Install PostgreSQL: PostgreSQL.
    \item \textbf{Step 5:} Navigate to the folder where you want to install the project and open the terminal.
    \item \textbf{Step 6:} Clone the project repository:
\end{itemize}
\begin{quote}\ttfamily
 git clone https://github.com/phattai-gif/CNPM-SUM26.git
\end{quote}
\begin{itemize}[leftmargin=1.5em, nosep]
    \item \textbf{Step 7:} Navigate to the project folder:
\end{itemize}
\begin{quote}\ttfamily
 cd CNPM-SUM26
\end{quote}
\begin{itemize}[leftmargin=1.5em, nosep]
    \item \textbf{Step 8:} Create and activate the Python virtual environment:
\end{itemize}
\begin{quote}\ttfamily
 python -m venv .venv\\
 .venv\textbackslash{}Scripts\textbackslash{}activate
\end{quote}
\begin{itemize}[leftmargin=1.5em, nosep]
    \item \textbf{Step 9:} Install the required packages:
\end{itemize}
\begin{quote}\ttfamily
 pip install -r requirements.txt
\end{quote}
\begin{itemize}[leftmargin=1.5em, nosep]
    \item \textbf{Step 10:} Configure the database connection in the \texttt{.env} file according to the project configuration.
    \item \textbf{Step 11:} Initialize the database and demo data:
\end{itemize}
\begin{quote}\ttfamily
 python src/scripts/db.py init
\end{quote}
\begin{itemize}[leftmargin=1.5em, nosep]
    \item \textbf{Step 12:} Run the backend application:
\end{itemize}
\begin{quote}\ttfamily
 python src/app.py
\end{quote}

\paragraph*{2.2.2 Frontend}
\addcontentsline{toc}{paragraph}{\protect\numberline{2.2.2}Frontend}

\begin{itemize}[leftmargin=1.5em, nosep]
    \item \textbf{Step 1:} Open the project using Visual Studio Code.
    \item \textbf{Step 2:} Navigate to the frontend directory:
\end{itemize}
\begin{quote}\ttfamily
 cd frontend
\end{quote}
\begin{itemize}[leftmargin=1.5em, nosep]
    \item \textbf{Step 3:} Configure the API address in \texttt{static/api\_client.js}.
    \item \textbf{Step 4:} Start the backend application.
    \item \textbf{Step 5:} Open the web application using the configured local address in a web browser.
\end{itemize}
